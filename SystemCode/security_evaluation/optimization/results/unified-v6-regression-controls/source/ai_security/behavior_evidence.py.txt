"""Responsibilities: Bind model-selected violation witnesses to complete actual proposal data.
Implementation: Build immutable indexed spans of JSON leaves or complete free-form text chunks;
retain the original proposal unchanged and never include input evidence in selectable spans.
Related Modules: behavior_semantic sends path/offset locators without duplicate text and validates
the returned index against these locally retained exact spans.
Declaration Index:
- ProposalSpan: Immutable source path, offset and actual proposal text reference.
- _unique_object: Reject ambiguous JSON object keys when constructing a structured data view.
- _reject_constant: Reject nonfinite JSON constants for the optional structured proposal view.
- _append_text: Append every character as ordered bounded source spans, including empty values.
- _append_json: Traverse all JSON fields, arrays, scalars and empty containers with stable paths.
- proposal_spans: Build complete selectable proposal references without changing proposed behavior.
Variable Index:
- SPAN_CHARS: Maximum reference-view chunk size; no proposal characters are discarded.
"""

import json
from typing import Literal

from pydantic import Field

from shared.contracts.behavior import BehaviorRequest
from shared.contracts.security import SecurityModel

SPAN_CHARS = 512


class ProposalSpan(SecurityModel):
    """Functionality: Bind one model-selectable reference to actual proposed data.
    Logic: Store stable index, source section, JSON pointer, character offsets and immutable text.
    Constraints: Offsets apply to the indicated scalar/text value; references have no authority
    and cannot include evidence. Text is copied from the proposal, not generated by a model.
    """

    index: int = Field(ge=0)
    source: Literal["content", "arguments"]
    path: str
    start: int = Field(ge=0)
    end: int = Field(ge=0)
    text: str


def _unique_object(pairs):
    """Functionality: Recognize only unambiguous structured proposal views. Inputs: JSON pairs.
    Outputs: Unique-key dictionary or ValueError. Logic: Reject duplicate keys instead of choosing
    a value. Constraints: This is a view parser for free-form proposal data, never verdict parsing.
    """
    value = {}
    for key, item in pairs:
        if key in value:
            raise ValueError("ambiguous proposal JSON view")
        value[key] = item
    return value


def _append_text(spans, text, source, path):
    """Functionality: Add all characters of one actual proposal value. Inputs: Mutable result list,
    exact text, source and pointer. Outputs: Appended immutable spans. Logic: Partition at 512
    characters with explicit offsets; represent empty values once. Constraints: No normalization,
    dropping whitespace or truncation; Python Unicode character offsets are not UTF-8 byte offsets.
    """
    for start in range(0, max(len(text), 1), SPAN_CHARS):
        end = min(start + SPAN_CHARS, len(text))
        spans.append(
            ProposalSpan(
                index=len(spans),
                source=source,
                path=path,
                start=start,
                end=end,
                text=text[start:end],
            )
        )


def _reject_constant(value):
    """Functionality: Reject nonstandard JSON numeric constants. Inputs: Parsed constant token.
    Outputs: ValueError. Logic: JSON permits only finite numbers. Constraints: No token is logged;
    free-form original content is retained if it is not recognizable as strict JSON.
    """
    raise ValueError("nonfinite proposal JSON constant")


def _append_json(spans, value, source, path):
    """Functionality: Reference every actual JSON value. Inputs: Span list, JSON value/source/path.
    Outputs: Appended spans. Logic: Preserve object order and array positions; encode pointer keys;
    retain empty containers, strings and canonical nonstring scalars. Constraints: Entire original
    serialized proposal remains separately available; this view adds no hidden or inferred facts.
    """
    if isinstance(value, dict) and value:
        for key, item in value.items():
            segment = key.replace("~", "~0").replace("/", "~1")
            _append_json(spans, item, source, path + "/" + segment)
    elif isinstance(value, list) and value:
        for index, item in enumerate(value):
            _append_json(spans, item, source, path + "/" + str(index))
    else:
        text = (
            value
            if isinstance(value, str)
            else json.dumps(value, ensure_ascii=False, allow_nan=False, separators=(",", ":"))
        )
        _append_text(spans, text, source, path)


def proposal_spans(request: BehaviorRequest) -> tuple[ProposalSpan, ...]:
    """Functionality: Build server-bound violation references from the complete actual proposal.
    Inputs: Strict request. Outputs: Immutable spans for content and arguments only.
    Logic: Add a structured leaf view when the whole content is unambiguous JSON; otherwise retain
    its complete original free-form text. Traverse all actual arguments. Constraints: Content is
    contractually free-form, so unsuccessful JSON recognition is a text view, not a repaired JSON
    verdict or replacement compliance implementation. No inputs/labels are selectable; raw proposal
    is unchanged and sent in full. Only span locators are transmitted; exact texts remain local.
    Selecting a real span proves attribution, not semantic violation.
    """
    spans = []
    if request.proposal.content is not None:
        text = request.proposal.content.text
        try:
            value = json.loads(
                text, object_pairs_hook=_unique_object, parse_constant=_reject_constant
            )
        except (ValueError, RecursionError):
            _append_text(spans, text, "content", "")
        else:
            _append_json(spans, value, "content", "")
    _append_json(spans, request.proposal.arguments, "arguments", "")
    return tuple(spans)
